<H1> Data Approval Pengajuan Pinjaman </H1>
<li> customer_id --> Id Customer </li>
<li> age --> Umur Customer </li>
<li> occupation_status --> status pekerjaan </li>
<li> years_employed --> lama bekerja </li>
<li> annual_income --> jumlah pemasukan tahunan (dalam dolar) </li>
<li> credit_score --> skor kredit, semakin tinggi semakin mudah di approve </li>
<li> credit_history_years --> catatan kredit sebelumnya </li>
<li> savings_assets --> aset tabungan </li>
<li> current_debt --> utang saat ini </li>
<li> defaults_on_file --> gagal bayar </li>
<li> delinquencies_last_2yrs --> catatan kol 2 tahun terakhir </li>
<li> derogatory_marks --> penanda pinjaman buruk </li>
<li> product_type --> tipe produk kredit </li>
<li> loan_intent --> tujuan pengajuan kredit </li>
<li> loan_amount --> jumlah pinjaman </li>
<li> interest_rate --> suku bunga pinjaman </li>
<li> debt_to_income_ratio --> rasio pemasukan dan total utang </li>
<li> loan_to_income_ratio --> rasio pemasukan dan pinjaman </li>
<li> payment_to_income_ratio --> rasio pemasukan dan cicilan </li>
<li> loan_status --> status approval pengajuan pinjaman, 0 jika ditolak dan 1 jika diterima </li>
